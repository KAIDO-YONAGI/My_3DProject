# 09 客户端显示

创建 `Unity/MvpView.cs`。客户端输入使用工程启用的 Unity Input System 输入系统。按键边沿先缓存，在下次 30Hz 输入发送时消费。客户端通过自己拥有的 Mirror 玩家身份发送输入，服务器再次检查归属与状态。

客户端同一播放 Tick 采样全部对象，完整基线决定当前几何。网络对象到达时，按对应业务身份建立形状。主机模式中的本地画面直接显示服务器物理对象，客户端的界面与输入继续使用相同消息链。

```csharp file=Unity/MvpView.cs
using System;
using System.Collections.Generic;
using System.IO;
using System.Linq;
using Mirror;
using UnityEngine;
using UnityEngine.InputSystem;

namespace Mvp
{
    public sealed class MvpView : MonoBehaviour
    {
        public MvpNetwork Manager;
        Replica replica = new Replica();
        PlaybackClock clock = new PlaybackClock();
        readonly List<MvpExplosion> events = new List<MvpExplosion>();
        readonly HashSet<(uint Round, uint Event)> seen = new HashSet<(uint, uint)>();
        readonly Dictionary<uint, uint> built = new Dictionary<uint, uint>();
        readonly Dictionary<uint, Transform> visuals = new Dictionary<uint, Transform>();
        Transform fixedRoot;
        uint fixedVersion = uint.MaxValue, visualRound;
        uint sequence;
        double nextInput, nextRecovery;
        byte buttons;
        public int InvalidPackets { get; private set; }
        public long BytesReceived { get; private set; }
        WorldState displayed;

        public void ResetClient()
        {
            replica = new Replica(); clock = new PlaybackClock(); events.Clear(); seen.Clear(); built.Clear();
            foreach (var root in visuals.Values) if (root != null) Destroy(root.gameObject);
            visuals.Clear();
            if (fixedRoot != null) Destroy(fixedRoot.gameObject);
            fixedRoot = null; fixedVersion = uint.MaxValue; visualRound = 0;
            sequence = 0; nextInput = 0; nextRecovery = 0; buttons = 0;
            displayed = null; InvalidPackets = 0; BytesReceived = 0;
        }

        public void ReceiveBaseline(byte[] bytes)
        {
            try
            {
                var w = replica.ReceiveBaseline(bytes);
                if (w == null) return;
                BytesReceived += bytes.Length;
                clock.Observe(w.Round, w.Tick, Time.realtimeSinceStartupAsDouble);
            }
            catch (Exception e) when (e is IOException || e is InvalidDataException)
            { ++InvalidPackets; RequestRecovery(); }
        }

        public void ReceiveMotion(byte[] bytes)
        {
            try
            {
                replica.ReceiveMotion(bytes);
                var h = Wire.Header(bytes);
                BytesReceived += bytes.Length;
                clock.Observe(h.Round, h.Tick, Time.realtimeSinceStartupAsDouble);
            }
            catch (Exception e) when (e is IOException || e is InvalidDataException)
            { ++InvalidPackets; RequestRecovery(); }
        }

        public void ReceiveExplosion(MvpExplosion e)
        {
            if (e.Round < replica.Round || !seen.Add((e.Round, e.Event))) return;
            events.Add(e);
            while (events.Count > 128) events.RemoveAt(0);
            if (seen.Count > 256)
                foreach (var key in seen.OrderBy(k => k.Round).ThenBy(k => k.Event).Take(128).ToArray())
                    seen.Remove(key);
        }

        void RequestRecovery()
        {
            double now = Time.realtimeSinceStartupAsDouble;
            if (!NetworkClient.ready || now < nextRecovery) return;
            nextRecovery = now + .5;
            NetworkClient.Send(new MvpRecover { Round = replica.Round }, Channels.Reliable);
        }

        void Update()
        {
            if (!NetworkClient.ready) return;
            double now = Time.realtimeSinceStartupAsDouble;
            SendInput(now);
            if (replica.NeedsBaseline) RequestRecovery();
            double tick = clock.Advance(now, Time.unscaledDeltaTime);
            displayed = replica.Render(tick);
            if (displayed != null && !NetworkServer.active) DrawWorld(displayed);
            foreach (var e in events.Where(e => e.Round == replica.Round && e.Tick <= tick).ToArray())
            {
                var flash = MvpWorld.Shape(PrimitiveType.Sphere, null, e.Position,
                                          Vector3.one * 2.5f, Color.red, false);
                Destroy(flash, .2f); events.Remove(e);
            }
            events.RemoveAll(e => e.Round < replica.Round);
        }

        void SendInput(double now)
        {
            var keyboard = Keyboard.current;
            if (keyboard == null || NetworkClient.localPlayer == null) return;
            if (keyboard.spaceKey.wasPressedThisFrame) buttons |= 1;
            if (keyboard.eKey.wasPressedThisFrame) buttons |= 2;
            if (now < nextInput) return;
            nextInput = now + Rules.Step;
            float x = (keyboard.dKey.isPressed ? 1 : 0) - (keyboard.aKey.isPressed ? 1 : 0);
            float z = (keyboard.wKey.isPressed ? 1 : 0) - (keyboard.sKey.isPressed ? 1 : 0);
            NetworkClient.Send(new MvpInput { Round = replica.Round, Sequence = ++sequence,
                X = x, Z = z, Buttons = buttons }, Channels.Reliable);
            buttons = 0;
        }

        void DrawWorld(WorldState w)
        {
            if (visualRound != w.Round)
            {
                visualRound = w.Round; fixedVersion = uint.MaxValue; built.Clear();
                foreach (var root in visuals.Values) if (root != null) Destroy(root.gameObject);
                visuals.Clear();
            }
            if (fixedRoot == null) fixedRoot = new GameObject("ClientFixedStructure").transform;
            if (fixedVersion != w.Version)
            {
                MvpWorld.ClearShapes(fixedRoot);
                var graph = new Structure();
                foreach (var block in w.Blocks.Where(b => b.Parent == 0))
                    MvpWorld.Shape(PrimitiveType.Cube, fixedRoot, MvpWorld.U(block.Local),
                        new Vector3(1.45f, .5f, 1.45f), graph.Anchor[block.Node] ? Color.green : Color.gray, false);
                fixedVersion = w.Version;
            }
            foreach (var s in w.Bodies)
            {
                if (!visuals.TryGetValue(s.Id, out var root))
                {
                    if (!MvpBody.ClientBodies.TryGetValue(s.Id, out var body) ||
                        body.Round != w.Round || body.Generation != s.Generation) continue;
                    root = new GameObject($"View-{s.Id}").transform;
                    visuals.Add(s.Id, root);
                }
                // 网络身份先销毁时，显示根继续存在到播放时间对应的拓扑生效。
                root.SetPositionAndRotation(MvpWorld.U(s.Position), MvpWorld.U(s.Rotation));
                if (built.TryGetValue(s.Id, out uint version) && version == w.Version) continue;
                MvpWorld.ClearShapes(root);
                if (s.Kind == BodyKind.Fragment)
                    foreach (var block in w.Blocks.Where(b => b.Parent == s.Id))
                        MvpWorld.Shape(PrimitiveType.Cube, root, MvpWorld.U(block.Local),
                            new Vector3(1.45f, .5f, 1.45f), Color.yellow, false);
                else
                    MvpWorld.Shape(s.Kind == BodyKind.Player ? PrimitiveType.Capsule : PrimitiveType.Sphere,
                        root, s.Kind == BodyKind.Player ? Vector3.up : Vector3.zero,
                        s.Kind == BodyKind.Player ? Vector3.one : Vector3.one * .45f,
                        s.Kind == BodyKind.Player ? Color.cyan : Color.red, false);
                built[s.Id] = w.Version;
            }
            foreach (uint id in built.Keys.Where(id => !w.Bodies.Any(s => s.Id == id)).ToArray())
            {
                built.Remove(id); Destroy(visuals[id].gameObject); visuals.Remove(id);
            }
        }

        void OnGUI()
        {
            if (Application.isBatchMode) return;
            if (!NetworkClient.active && !NetworkServer.active)
            {
                Manager.networkAddress = GUI.TextField(new Rect(10, 10, 180, 25), Manager.networkAddress);
                if (GUI.Button(new Rect(200, 10, 90, 25), "Client")) Manager.StartClient();
                if (GUI.Button(new Rect(300, 10, 90, 25), "Host")) Manager.StartHost();
                if (GUI.Button(new Rect(400, 10, 90, 25), "Server")) Manager.StartServer();
                return;
            }
            GUILayout.BeginArea(new Rect(10, 10, 400, 260));
            GUILayout.Label($"Round {replica.Round} / Tick {clock.Tick:F1}");
            GUILayout.Label($"Receive {BytesReceived} B / Invalid {InvalidPackets}");
            if (NetworkServer.active)
                GUILayout.Label($"Send {Manager.World.BytesSent} B / Step {Manager.World.LastStepMs:F2} ms");
            if (displayed != null)
            {
                GUILayout.Label($"Bodies {displayed.Bodies.Length} / Fragments " +
                    displayed.Bodies.Count(s => s.Kind == BodyKind.Fragment));
                foreach (var s in displayed.Bodies.Where(s => s.Kind == BodyKind.Player))
                    GUILayout.Label($"Player {s.Id}: HP {s.Health} / Score {s.Score}");
            }
            GUILayout.EndArea();
        }
    }
}
```

本章检查：快速按下并松开跳跃仍然发出一次操作；爆炸事件重复到达时播放一次；完整基线先到、对象稍后创建时显示恢复；客户端形状的碰撞器关闭，共享物理由服务器执行。
