# 07 角色与动态支撑

创建 `Unity/MvpWorld.Players.cs`。角色使用服务器的 `CharacterController` 胶囊控制器。一次 Tick 合并支撑位移、主动移动、重力和击退，调用一次 `Move`。输入消息中的跳跃和投掷位在 Tick 消费后清零，持续方向超过六 Tick 未更新时归零。

动态支撑保存碎片编号、代次、局部接触点和上次世界接触点。下个 Tick 把同一局部点转换到新世界位置，取得平移与旋转共同产生的位移。跳离时使用刚体接触点速度，包含角速度贡献。

```csharp file=Unity/MvpWorld.Players.cs
using Mirror;
using UnityEngine;

namespace Mvp
{
    public sealed partial class MvpWorld
    {
        sealed class Player
        {
            public MvpBody Body;
            public CharacterController Controller;
            public InputGate Gate = new InputGate();
            public Vector3 Move, Velocity, External;
            public byte Buttons;
            public uint LastInput, LastThrow, LastImpact;
            public short Health = 100, Score;
            public uint Support;
            public ushort SupportGeneration;
            public Vector3 LocalPoint, PreviousPoint;
            public float Vertical;
        }

        public void AddPlayer(MvpBody body)
        {
            body.Id = Allocate(); body.Round = Round; body.Kind = BodyKind.Player;
            body.transform.position = SpawnPoint(players.Count);
            var cc = body.gameObject.AddComponent<CharacterController>();
            cc.center = new Vector3(0, 1, 0); cc.height = 2; cc.radius = .35f;
            cc.skinWidth = .03f; cc.stepOffset = .2f; cc.minMoveDistance = 0;
            Shape(PrimitiveType.Capsule, body.transform, Vector3.up, Vector3.one,
                  players.Count % 2 == 0 ? Color.cyan : Color.magenta, false);
            bodies.Add(body.Id, body);
            players.Add(body.Id, new Player { Body = body, Controller = cc });
            ++Version; dirty = true;
        }

        static Vector3 SpawnPoint(int index) => new Vector3(index % 2 == 0 ? -4.5f : 4.5f, 3, 0);

        public void RemovePlayer(MvpBody body)
        {
            if (body == null) return;
            players.Remove(body.Id); bodies.Remove(body.Id);
            ++Version; dirty = true;
        }

        public void ReceiveInput(uint id, MvpInput input)
        {
            if (!players.TryGetValue(id, out var p) || p.Health <= 0 ||
                !p.Gate.Accept(input.Round, Round, input.Sequence, input.X, input.Z, input.Buttons, Tick))
                return;
            p.Move = Vector3.ClampMagnitude(new Vector3(input.X, 0, input.Z), 1);
            p.Buttons |= input.Buttons; p.LastInput = Tick;
        }

        void MovePlayer(Player p)
        {
            if (p.Health <= 0) { p.Buttons = 0; return; }
            var t = p.Body.transform;
            Vector3 carry = Vector3.zero, pointVelocity = Vector3.zero;
            if (p.Support != 0 && bodies.TryGetValue(p.Support, out var support) &&
                support.Generation == p.SupportGeneration && support.Kind == BodyKind.Fragment)
            {
                carry = support.transform.TransformPoint(p.LocalPoint) - p.PreviousPoint;
                pointVelocity = support.GetComponent<Rigidbody>().GetPointVelocity(t.position + carry);
            }
            else p.Support = 0;

            bool grounded = Ground(t.position + carry, out var hit);
            if (grounded && p.Vertical < 0) p.Vertical = -1;
            bool jump = (p.Buttons & 1) != 0 && grounded;
            if (jump)
            {
                p.Vertical = 8 + pointVelocity.y;
                p.External += new Vector3(pointVelocity.x, 0, pointVelocity.z);
                p.Support = 0;
            }
            p.Vertical -= 20 * Rules.Step;
            Vector3 direction = Tick - p.LastInput <= 6 ? p.Move : Vector3.zero;
            if (direction.sqrMagnitude > .01f) t.rotation = Quaternion.LookRotation(direction);
            p.Velocity = direction * 5 + p.External + Vector3.up * p.Vertical;
            var flags = p.Controller.Move(carry + p.Velocity * Rules.Step);
            if ((flags & CollisionFlags.Above) != 0 && p.Vertical > 0) p.Vertical = 0;
            p.External = Vector3.MoveTowards(p.External, Vector3.zero, 3 * Rules.Step);

            p.Support = 0;
            if (!jump && p.Vertical <= 0 && Ground(t.position, out hit))
            {
                p.Vertical = -1;
                var body = hit.collider.GetComponentInParent<MvpBody>();
                if (body != null && body.Kind == BodyKind.Fragment)
                {
                    p.Support = body.Id; p.SupportGeneration = body.Generation;
                    p.LocalPoint = body.transform.InverseTransformPoint(hit.point);
                    p.PreviousPoint = body.transform.TransformPoint(p.LocalPoint);
                }
            }
            if ((p.Buttons & 2) != 0) ThrowBomb(p);
            p.Buttons = 0;
        }

        bool Ground(Vector3 feet, out RaycastHit hit)
        {
            if (!Physics.SphereCast(feet + Vector3.up * .3f, .25f, Vector3.down,
                                    out hit, .25f, ~0, QueryTriggerInteraction.Ignore)) return false;
            var body = hit.collider.GetComponentInParent<MvpBody>();
            return body == null || body.Kind == BodyKind.Fragment;
        }

        void Damage(Player p, int amount, Vector3 impulse)
        {
            if (p.Health <= 0 || endTick != 0) return;
            p.Health = (short)Mathf.Max(0, p.Health - amount);
            p.External += new Vector3(impulse.x, 0, impulse.z);
            p.Vertical += Mathf.Max(0, impulse.y);
            if (p.Health == 0)
            {
                p.Controller.enabled = false;
                p.Support = 0; p.Body.transform.position = new Vector3(0, -20, 0);
            }
        }

        public void CharacterImpact(MvpBody body, ControllerColliderHit hit)
        {
            if (!players.TryGetValue(body.Id, out var p) || hit.rigidbody == null ||
                Tick - p.LastImpact < 6) return;
            var other = hit.rigidbody.GetComponent<MvpBody>();
            if (other == null || other.Kind != BodyKind.Fragment || other.Id == p.Support) return;
            Vector3 relative = hit.rigidbody.GetPointVelocity(hit.point) - p.Velocity;
            // 使用沿接触法线靠近角色的速度，站立和沿表面滑动保持接触。
            float speed = Mathf.Max(0, Vector3.Dot(relative, hit.normal));
            if (speed < 6) return;
            p.LastImpact = Tick;
            Damage(p, Mathf.Min(35, (int)(speed * 2)), relative.normalized * Mathf.Min(8, speed));
        }
    }
}
```

第 05 章的 `MvpBody.OnControllerColliderHit` 把胶囊接触交给本章的 `CharacterImpact`。碎片撞击和爆炸共同使用 `Damage`，淘汰与击退由同一入口处理。

支撑销毁或代次变化时，支撑编号在下次 Tick 清空。支撑切换在本次移动后的地面探测中确定。保持接触点在碎片的局部空间，使碎片旋转时角色也获得相应位移。

本章检查：两端出生点落在平台内；站在旋转碎片上时脚下位移连续；跳离后获得碎片的接触点速度；支撑消失后角色进入下落；每个输入序号只进入一次服务器消费过程。
